cmake_minimum_required(VERSION 3.16)
project(indi_host_checks LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../..")
add_executable(indi_core_test test_core.cpp "${ROOT}/src/lib/indi/IndiControl.cpp")
target_include_directories(indi_core_test PRIVATE "${ROOT}/src/lib/indi")
target_compile_options(indi_core_test PRIVATE -Wall -Wextra -Werror -pedantic -fno-exceptions -fno-rtti)
option(INDI_SANITIZERS "Enable host address and undefined behavior sanitizers" ON)
if(INDI_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(indi_core_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(indi_core_test PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME indi_core COMMAND indi_core_test)
